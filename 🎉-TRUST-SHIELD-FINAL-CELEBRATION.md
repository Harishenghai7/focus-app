### **useTrustShield() Hook**

```jsx
import useTrustShield from './hooks/useTrustShield';

function MyComponent() {
  const {
    // Core Data
    trustScore,           // 0-100 trust score
    verificationLevel,    // Current level
    trustStatus,          // Full status object
    
    // Restrictions & Limits
    restrictions,         // Account restrictions
    rateLimits,          // Current rate limits
    badges,              // Earned badges
    
    // State
    isLoading,           // Loading indicator
    error,               // Error message
    
    // Helpers
    isVerified,          // Quick verified check
    isTrusted,           // Quick trust check
    hasRestrictions,     // Quick restriction check
    
    // Functions
    canPerform,          // Check action permission
    updateTrust,         // Update trust score
    refreshStatus        // Refresh data
  } = useTrustShield();
  
  return (
    <div>
      Trust Score: {trustScore}
      Level: {verificationLevel}
    </div>
  );
}
```

## 🎨 EXAMPLE USE CASES

### **1. Post Creation with Trust Check**
```jsx
function CreatePost() {
  const { canPerform } = useTrustShield();
  
  const handleSubmit = async () => {
    const permission = await canPerform('post');
    
    if (!permission.allowed) {
      showError(permission.reason);
      return;
    }
    
    await createPost();
  };
  
  return <button onClick={handleSubmit}>Post</button>;
}
```

### **2. Trust Score Display**
```jsx
function TrustBadge() {
  const { trustScore, verificationLevel } = useTrustShield();
  
  return (
    <div className={`badge ${verificationLevel}`}>
      <span className="score">{trustScore}</span>
      <span className="level">{verificationLevel}</span>
    </div>
  );
}
```

### **3. Feature Gate by Trust Level**
```jsx
function PremiumFeatures() {
  const { isTrusted } = useTrustShield();
  
  if (!isTrusted) {
    return <UpgradeTrustPrompt />;
  }
  
  return <PremiumContent />;
}
```

### **4. Rate Limit Display**
```jsx
function ActionLimitIndicator() {
  const { rateLimits, canPerform } = useTrustShield();
  const [remaining, setRemaining] = useState(0);
  
  useEffect(() => {
    canPerform('post').then(p => setRemaining(p.remaining));
  }, []);
  
  return (
    <div>
      Posts remaining: {remaining} / {rateLimits.posts_per_hour}
    </div>
  );
}
```
