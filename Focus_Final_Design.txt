import React from 'react';
import { useNavigate } from 'react-router-dom';
import { useAuth } from '../../hooks/useAuth';
import { useNotifications } from '../../hooks/useNotifications';
import ThemeToggle from '../ThemeToggle';
import styles from './MobileHeader.module.css';

const MobileHeader = ({ pageTitle, showBackButton = false }) => {
  const navigate = useNavigate();
  const { user } = useAuth();
  const { unreadMessages, missedCalls, unreadNotifications } = useNotifications();

  return (
    <header className={styles.header}>
      {/* Left Section */}
      <div className={styles.left}>
        {showBackButton ? (
          <button className={styles.backButton} onClick={() => navigate(-1)}>
            <svg width="24" height="24" viewBox="0 0 24 24" fill="none">
              <path d="M15 18L9 12L15 6" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"/>
            </svg>
          </button>
        ) : (
          <div className={styles.logo}>
            Focus
            <span className={styles.logoDot}>.</span>
          </div>
        )}
      </div>

      {/* Center Section (Page Title - Optional) */}
      {pageTitle && (
        <div className={styles.center}>
          <h1 className={styles.pageTitle}>{pageTitle}</h1>
        </div>
      )}

      {/* Right Section */}
      <div className={styles.right}>
        <button 
          className={styles.iconButton}
          onClick={() => navigate('/messages')}
        >
          <svg width="24" height="24" viewBox="0 0 24 24" fill="none">
            <path d="M21 15C21 15.5304 20.7893 16.0391 20.4142 16.4142C20.0391 16.7893 19.5304 17 19 17H7L3 21V5C3 4.46957 3.21071 3.96086 3.58579 3.58579C3.96086 3.21071 4.46957 3 5 3H19C19.5304 3 20.0391 3.21071 20.4142 3.58579C20.7893 3.96086 21 4.46957 21 5V15Z" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"/>
          </svg>
          {unreadMessages > 0 && (
            <span className={styles.badge}>{unreadMessages}</span>
          )}
        </button>

        <button 
          className={styles.iconButton}
          onClick={() => navigate('/calls')}
        >
          <svg width="24" height="24" viewBox="0 0 24 24" fill="none">
            <path d="M22 16.92V19.92C22 20.4728 21.5523 20.92 21 20.92H3C2.44772 20.92 2 20.4728 2 19.92V16.92C2 16.3677 2.44772 15.92 3 15.92H21C21.5523 15.92 22 16.3677 22 16.92Z" stroke="currentColor" strokeWidth="2"/>
          </svg>
          {missedCalls > 0 && (
            <span className={styles.badge}>{missedCalls}</span>
          )}
        </button>

        <button 
          className={styles.iconButton}
          onClick={() => navigate('/notifications')}
        >
          <svg width="24" height="24" viewBox="0 0 24 24" fill="none">
            <path d="M18 8C18 6.4087 17.3679 4.88258 16.2426 3.75736C15.1174 2.63214 13.5913 2 12 2C10.4087 2 8.88258 2.63214 7.75736 3.75736C6.63214 4.88258 6 6.4087 6 8C6 15 3 17 3 17H21C21 17 18 15 18 8Z" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"/>
            <path d="M13.73 21C13.5542 21.3031 13.3019 21.5547 12.9982 21.7295C12.6946 21.9044 12.3504 21.9965 12 21.9965C11.6496 21.9965 11.3054 21.9044 11.0018 21.7295C10.6982 21.5547 10.4458 21.3031 10.27 21" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"/>
          </svg>
          {unreadNotifications > 0 && (
            <span className={styles.badge}>{unreadNotifications}</span>
          )}
        </button>

        <ThemeToggle />
      </div>
    </header>
  );
};

export default MobileHeader;
